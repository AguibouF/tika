@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_available_5_0_Test {

    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(mockStream, 10);
    }

    @Test
    public void testAvailable() throws IOException {
        when(mockStream.available()).thenReturn(15);

        assertEquals(5, result);

    }

        when(mockStream.available()).thenReturn(-5);

        assertEquals(-5, result);
        verify(mockStream, times(1)).available();
    }

    @Test
    public void testAvailableWithBufferedData() throws IOException {
        when(mockStream.available()).thenReturn(20);
        lookaheadInputStream.read();
        lookaheadInputStream.read();
        int result = lookaheadInputStream.available();
        assertEquals(18, result);
        verify(mockStream, times(1)).available();
    }
}
