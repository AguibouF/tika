package org.apache.tika.io;

import org.apache.tika.io.LookaheadInputStream;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.IOException;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_read_2_0_Test {

    @Mock
    private InputStream mockStream;

    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(mockStream, 10);
    }

    @Test
    public void testRead() throws IOException {
        when(mockStream.read()).thenReturn(100);
        when(mockStream.available()).thenReturn(1);
        int result = lookaheadInputStream.read();
        assertEquals(100, result);
        verify(mockStream, times(1)).read();
        verify(mockStream, times(1)).available();
    }

    @Test
    public void testReadEndOfStream() throws IOException {
        when(mockStream.read()).thenReturn(-1);
        int result = lookaheadInputStream.read();
        assertEquals(-1, result);
        verify(mockStream, times(1)).read();
    }

    @Test
    public void testReadIOException() throws IOException {
        when(mockStream.read()).thenThrow(new IOException("Mocked IOException"));
        assertThrows(IOException.class, () -> lookaheadInputStream.read());
        verify(mockStream, times(1)).read();
    }

    @Test
    public void testFill() throws IOException {
        when(mockStream.available()).thenReturn(0);
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(5);
        lookaheadInputStream.fill();
        verify(mockStream, times(1)).available();
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
    }

    @Test
    public void testFillEndOfStream() throws IOException {
        when(mockStream.available()).thenReturn(0);
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(-1);
        lookaheadInputStream.fill();
        verify(mockStream, times(1)).available();
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
    }

    @Test
    public void testFillIOException() throws IOException {
        when(mockStream.available()).thenReturn(0);
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenThrow(new IOException("Mocked IOException"));
        assertThrows(IOException.class, () -> lookaheadInputStream.fill());
        verify(mockStream, times(1)).available();
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
    }
}
