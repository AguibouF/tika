package org.apache.tika.io;

import org.apache.tika.io.LookaheadInputStream;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.IOException;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_skip_4_0_Test {

    @Mock
    private InputStream mockStream;

    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(mockStream, 10);
    }

    @Test
    public void testSkip() throws IOException {
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(5);
        long result = lookaheadInputStream.skip(3);
        assertEquals(3, result);
        assertEquals(3, lookaheadInputStream.position);
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
    }

    @Test
    public void testSkipMoreThanAvailable() throws IOException {
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(5);
        long result = lookaheadInputStream.skip(10);
        assertEquals(5, result);
        assertEquals(5, lookaheadInputStream.position);
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
    }

    @Test
    public void testSkipZero() throws IOException {
        long result = lookaheadInputStream.skip(0);
        assertEquals(0, result);
        assertEquals(0, lookaheadInputStream.position);
    }

    @Test
    public void testSkipNegative() throws IOException {
        long result = lookaheadInputStream.skip(-1);
        assertEquals(0, result);
        assertEquals(0, lookaheadInputStream.position);
    }

    @Test
    public void testSkipAfterClose() throws IOException {
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(5);
        lookaheadInputStream.close();
        long result = lookaheadInputStream.skip(3);
        assertEquals(0, result);
        assertEquals(0, lookaheadInputStream.position);
        verify(mockStream, never()).read(any(byte[].class), anyInt(), anyInt());
    }
}
